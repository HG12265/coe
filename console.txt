        private void LoadStudentPhotos(DataSet dspdf)
        {
            try
            {
                string logFile = Path.Combine(Application.StartupPath, "photo_debug_log.txt");
                StringBuilder sbLog = new StringBuilder();
                sbLog.AppendLine("=== PHOTO DIAGNOSTIC LOG " + DateTime.Now.ToString() + " ===");

                if (dspdf != null && dspdf.Tables.Contains("Student_Info"))
                {
                    sbLog.AppendLine("Student_Info table found. Total Rows: " + dspdf.Tables["Student_Info"].Rows.Count);
                    bool hasPhotoCol = dspdf.Tables["Student_Info"].Columns.Contains("PHOTO");
                    sbLog.AppendLine("PHOTO column in Student_Info table: " + hasPhotoCol);

                    foreach (DataRow dr in dspdf.Tables["Student_Info"].Rows)
                    {
                        try
                        {
                            string regNo = "";
                            if (dr.Table.Columns.Contains("REGNO") && dr["REGNO"] != DBNull.Value)
                                regNo = dr["REGNO"].ToString().Trim();
                            else if (dr.Table.Columns.Contains("RegNo") && dr["RegNo"] != DBNull.Value)
                                regNo = dr["RegNo"].ToString().Trim();
                            else if (dr.Table.Columns.Contains("Reg_No") && dr["Reg_No"] != DBNull.Value)
                                regNo = dr["Reg_No"].ToString().Trim();

                            sbLog.AppendLine("\nProcessing RegNo: '" + regNo + "'");

                            if (string.IsNullOrEmpty(regNo))
                            {
                                sbLog.AppendLine("  -> SKIP: RegNo is empty!");
                                continue;
                            }

                            string filePath = "G://DUIC//duic_image//" + regNo + ".jpg";
                            string filePath_jpeg = "G://DUIC//duic_image//" + regNo + ".jpeg";
                            string filePath_JPG = "G://DUIC//duic_image//" + regNo + ".JPG";
                            string noImagePath = "G://DUIC//duic_image//noimage.jpg";

                            byte[] NoImage = null;
                            string matchedPath = "";

                            if (File.Exists(filePath))
                            {
                                matchedPath = filePath;
                                using (FileStream fs = new FileStream(filePath, FileMode.Open, FileAccess.Read))
                                {
                                    NoImage = new byte[fs.Length];
                                    fs.Read(NoImage, 0, Convert.ToInt32(fs.Length));
                                }
                            }
                            else if (File.Exists(filePath_jpeg))
                            {
                                matchedPath = filePath_jpeg;
                                using (FileStream fs = new FileStream(filePath_jpeg, FileMode.Open, FileAccess.Read))
                                {
                                    NoImage = new byte[fs.Length];
                                    fs.Read(NoImage, 0, Convert.ToInt32(fs.Length));
                                }
                            }
                            else if (File.Exists(filePath_JPG))
                            {
                                matchedPath = filePath_JPG;
                                using (FileStream fs = new FileStream(filePath_JPG, FileMode.Open, FileAccess.Read))
                                {
                                    NoImage = new byte[fs.Length];
                                    fs.Read(NoImage, 0, Convert.ToInt32(fs.Length));
                                }
                            }
                            else if (File.Exists(noImagePath))
                            {
                                matchedPath = noImagePath + " (FALLBACK)";
                                using (FileStream fs = new FileStream(noImagePath, FileMode.Open, FileAccess.Read))
                                {
                                    NoImage = new byte[fs.Length];
                                    fs.Read(NoImage, 0, Convert.ToInt32(fs.Length));
                                }
                            }
                            else
                            {
                                sbLog.AppendLine("  -> CRITICAL: Neither student photo nor noimage.jpg exists!");
                            }

                            if (!string.IsNullOrEmpty(matchedPath))
                            {
                                sbLog.AppendLine("  -> SUCCESS: Loaded from '" + matchedPath + "' | Bytes: " + (NoImage != null ? NoImage.Length : 0));
                            }

                            if (hasPhotoCol)
                            {
                                dr["PHOTO"] = NoImage;
                                sbLog.AppendLine("  -> ASSIGNED dr['PHOTO'] successfully.");
                            }
                            else
                            {
                                sbLog.AppendLine("  -> ERROR: 'PHOTO' column missing in Student_Info table!");
                            }
                        }
                        catch (Exception ex)
                        {
                            sbLog.AppendLine("  -> EXCEPTION on row: " + ex.Message);
                        }
                    }
                }
                else
                {
                    sbLog.AppendLine("CRITICAL ERROR: dspdf is null or does not contain Student_Info table!");
                }

                File.WriteAllText(logFile, sbLog.ToString());
            }
            catch { }
        }
